1110:    * - Within grace period: 0.000 KD
1111:    * - Crossed grace period (1st hour): Base rate from DB (1 KD for Single, 2 KD for Double)
1112:    * - Subsequent hours after 1st hour: 50% of base rate per hour (0.500 KD for Single, 1.000 KD for Double)
1113:    */

1115:     const car = this.selectedCar();

1118:     if (!r) return 0;
1119: 
1120:     const start = new Date(r.start_time || r.startTime).getTime();
1121:     const elapsedMins = Math.max(0, Math.floor((this.currentTime.getTime() - start) / 60000));
1122:     const grace = Number(this.gracePeriodMinutes() || 10);
1123: 
1124:     // Rule 1: Within initial grace period -> 0 KD
1125:     if (elapsedMins <= grace) {
1126:       return 0.000;
1127:     }
1128: 
1129:     const baseRate = this.getCarBaseRate(car);
1130:     const extraRatePerHour = baseRate * 0.5;
1131:

1133:     if (elapsedMins <= 60 + grace) {
1134:       return baseRate;
1135:     }
1136: 
1137:     // Rule 3: After first hour crossed + grace period -> Base rate + extra hours charge
1138:     const extraMins = elapsedMins - 60 - grace;
1139:     const extraHours = Math.floor(extraMins / 60) + 1; // Ceil of additional hours
1140:     return baseRate + (extraHours * extraRatePerHour);
